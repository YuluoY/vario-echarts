### 在代码中使用

```typescript
import zhCN from '@/data/i18n/zh-CN.json';
import enUS from '@/data/i18n/en-US.json';

function getLabel(key: string, locale: string = 'zh-CN') {
  const i18n = locale === 'zh-CN' ? zhCN : enUS;
  return i18n[key] || key;
}

console.log(getLabel('title', 'zh-CN')); // "标题"
console.log(getLabel('title', 'en-US')); // "Title"
```

### 国际化文件生成

**translator.ts**:
```typescript
private saveI18nFiles() {
  const i18nDir = join(dirname(this.dictPath), 'i18n');
  mkdirSync(i18nDir, { recursive: true });
  
  // 中文
  const zhCN = this.dict;
  writeFileSync(join(i18nDir, 'zh-CN.json'), JSON.stringify(zhCN, null, 2));
  
  // 英文（驼峰转空格）
  const enUS: Record<string, string> = {};
  Object.keys(this.dict).forEach(en => {
    enUS[en] = en.replace(/([A-Z])/g, ' $1').trim();
  });
  writeFileSync(join(i18nDir, 'en-US.json'), JSON.stringify(enUS, null, 2));
}
```

## 🎯 使用场景

### 场景 1：调试配置
```
1. 打开 Studio → 预览页面
2. 选择一个选项（如 title）
3. 左侧切换到"状态数据"查看实时状态
4. 右侧操作组件，观察左侧数据变化
```

### 场景 2：多语言应用
```typescript
// 根据用户语言显示标签
const locale = navigator.language.startsWith('zh') ? 'zh-CN' : 'en-US';
const i18n = locale === 'zh-CN' ? zhCN : enUS;

schema.forEach(node => {
  if (node.props?.placeholder) {
    node.props.placeholder = i18n[node.props.placeholder];
  }
});
```

### 场景 3：文档生成
```typescript
// 从 i18n 文件生成字段文档
Object.entries(zhCN).forEach(([en, zh]) => {
  console.log(`- **${en}**: ${zh} (${enUS[en]})`);
});

// 输出：
// - **title**: 标题 (Title)
// - **show**: 显示 (Show)
```
